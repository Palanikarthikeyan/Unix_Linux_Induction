Write a shell script
-----------------------
step 1:  initialize a pin number  (ex: pin=1234)
|
step 2: use while loop -limit is 3
	     |
	     | -> read a pinNumber from <STDIN>
	     | -> test a pinNumber with existing pin
	     | ->   display - input is success - exit from loop(break)

step 3: pin is blocked if all 3 inputs are failed.


pin=1234
count=0
while [ $count -lt 3 ]
do
     count=`expr $count + 1`
     read -p "Enter a pin:" p
     if [ $p -eq $pin ]
     then
	   echo "Success - pin is matched - $count"
	   break
     fi
done

if [ $p -ne $pin ]
then
    echo "Sorry your pin is blocked"
fi
